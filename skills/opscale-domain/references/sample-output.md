## 1. Model — `src/Models/Application.php`

```php
namespace Acme\Loans\Models;

use Illuminate\Database\Eloquent\Concerns\HasUlids;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Acme\Loans\Models\Enums\ApplicationStatus;
use Acme\Loans\Models\Repositories\ApplicationRepository;
use Opscale\Validations\Validatable;

class Application extends Model
{
    use ApplicationRepository;
    use HasUlids;
    use Validatable;

    public static $validationRules = [
        'code'      => ['required', 'string', 'max:20'],
        'party_id'  => ['required', 'string', 'ulid'],
        'policy_id' => ['required', 'string', 'ulid'],
    ];

    protected $fillable = ['code', 'status', 'party_id', 'policy_id', 'current_cycle'];

    protected $casts = [
        'status'        => ApplicationStatus::class,
        'current_cycle' => 'integer',
    ];

    public function policy(): BelongsTo
    {
        return $this->belongsTo(Policy::class);
    }
}
```

## 2. Enum — `src/Models/Enums/ApplicationStatus.php`

```php
enum ApplicationStatus: string
{
    case Pending   = 'Pending';
    case KycReady  = 'KYC Ready';
    case Qualified = 'Qualified';
    case Approved  = 'Approved';
    case Active    = 'Active';
    case Rejected  = 'Rejected';

    public function canTransitionTo(self $status): bool
    {
        return match ($this) {
            self::Pending   => $status === self::KycReady,
            self::KycReady  => in_array($status, [self::Qualified, self::Rejected], true),
            self::Qualified => in_array($status, [self::Approved, self::Rejected], true),
            self::Approved  => $status === self::Active,
            self::Active, self::Rejected => false,
        };
    }
}
```

## 3. Repository trait — `src/Models/Repositories/ApplicationRepository.php`

```php
trait ApplicationRepository
{
    public static function bootApplicationRepository(): void
    {
        static::creating(function ($model): void {
            if (empty($model->code)) {
                $year = (int) date('Y');
                $seq  = static::query()->where('code', 'like', "APP-{$year}-%")->count() + 1;
                $model->code = sprintf('APP-%d-%05d', $year, $seq);
            }
        });
    }
}
```

### 5a. Enum — `tests/Unit/Enums/ApplicationStatusTest.php`

Exact values, a non-empty `label()` per case, the **complete** transition matrix,
and terminal states.

```php
<?php

use Acme\Loans\Models\Enums\ApplicationStatus;

describe('ApplicationStatus Enum', function () {
    it('has exactly the DBML case values', function () {
        expect(array_map(fn ($c) => $c->value, ApplicationStatus::cases()))
            ->toBe(['Pending', 'KYC Ready', 'Approved', 'Rejected', 'Funded', 'Closed']);
    });

    it('gives every case a non-empty label', function () {
        foreach (ApplicationStatus::cases() as $case) {
            expect($case->label())->toBeString()->not->toBe('');
        }
    });

    // Transcribed from ApplicationStatus::canTransitionTo() — every (from, to) pair.
    dataset('transitions', [
        [ApplicationStatus::Pending,  ApplicationStatus::KycReady, true],
        [ApplicationStatus::Pending,  ApplicationStatus::Approved, false],
        [ApplicationStatus::KycReady, ApplicationStatus::Approved, true],
        [ApplicationStatus::KycReady, ApplicationStatus::Rejected, true],
        [ApplicationStatus::Approved, ApplicationStatus::Funded,   true],
        [ApplicationStatus::Rejected, ApplicationStatus::Approved, false],
        // ... one row per (from, to) pair in cases()^2, each with its expected bool
    ]);

    it('enforces the exact transition matrix', function ($from, $to, $allowed) {
        expect($from->canTransitionTo($to))->toBe($allowed);
    })->with('transitions');

    it('lets no transition leave a terminal state', function () {
        foreach ([ApplicationStatus::Rejected, ApplicationStatus::Closed] as $terminal) {
            foreach (ApplicationStatus::cases() as $to) {
                expect($terminal->canTransitionTo($to))->toBeFalse();
            }
        }
    });
});
```

### 5d. Repository — `tests/Unit/Repositories/ApplicationRepositoryTest.php`

Scopes include-and-exclude, plus the transition guard positive/negative/terminal.

```php
<?php

use Acme\Loans\Models\Application;
use Acme\Loans\Models\Enums\ApplicationStatus;
use Symfony\Component\HttpKernel\Exception\HttpException;

describe('ApplicationRepository scopes', function () {
    it('scopePending includes pending and excludes the rest', function () {
        $pending = Application::factory()->create(['status' => ApplicationStatus::Pending]);
        $funded  = Application::factory()->create(['status' => ApplicationStatus::Funded]);
        $ids = Application::pending()->pluck('id');
        expect($ids)->toContain($pending->id)->not->toContain($funded->id);
    });
});

describe('ApplicationRepository transition guard', function () {
    it('persists a valid transition', function () {
        $app = Application::factory()->create(['status' => ApplicationStatus::Pending]);
        $app->update(['status' => ApplicationStatus::KycReady]);
        expect($app->fresh()->status)->toBe(ApplicationStatus::KycReady);
    });

    it('blocks an invalid transition and does not persist it', function () {
        $app = Application::factory()->create(['status' => ApplicationStatus::Pending]);
        expect(fn () => $app->update(['status' => ApplicationStatus::Funded]))->toThrow(HttpException::class);
        expect($app->fresh()->status)->toBe(ApplicationStatus::Pending);
    });

    it('blocks any transition out of a terminal state', function () {
        $app = Application::factory()->create(['status' => ApplicationStatus::Closed]);
        expect(fn () => $app->update(['status' => ApplicationStatus::Approved]))->toThrow(HttpException::class);
    });
});
```

> single test proving a normal `update()` is not blocked.
